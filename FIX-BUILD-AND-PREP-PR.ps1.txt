param(
  [string]$RepoRoot = ".",
  [string]$FrontendDir = "frontend"
)

Write-Host "=== Confusion-AI: Fix build + prep branch for PR ===" -ForegroundColor Cyan

function Write-Utf8NoBom {
  param([string]$Path, [string]$Content)
  $utf8NoBom = New-Object System.Text.UTF8Encoding($false)
  [System.IO.File]::WriteAllText($Path, $Content, $utf8NoBom)
}

try { $root = (Resolve-Path $RepoRoot -ErrorAction Stop).Path } catch { Write-Error "Repo root not found"; exit 1 }
Set-Location $root
Write-Host "Repo root: $root" -ForegroundColor Yellow

$frontendPath = Join-Path $root $FrontendDir
if (-not (Test-Path $frontendPath)) { Write-Error "Frontend folder not found: $frontendPath"; exit 1 }

# --- 1) Fix the App.tsx syntax error ---
$appPath = Join-Path $frontendPath "src\App.tsx"
if (-not (Test-Path $appPath)) { Write-Error "frontend/src/App.tsx not found"; exit 1 }

$ts = Get-Date -Format "yyyyMMdd_HHmmss"
Copy-Item $appPath "$appPath.bak-$ts" -Force

$app = Get-Content $appPath -Raw

# Replace the broken setError(Free plan: ...)
$pattern = '(?m)^(\s*)setError\(\s*Free\s+plan:.*\);\s*$'
$replacement = '$1setError(`Free plan: ${FREE_DAILY_LIMIT} analyses per day. Upgrade to Pro for unlimited.`);'

$fixed = [regex]::Replace($app, $pattern, $replacement)

if ($fixed -eq $app) {
  # Fallback: catch any variation that starts with setError(Free plan
  $pattern2 = '(?m)^(\s*)setError\(\s*Free\s+plan.*\);\s*$'
  $fixed = [regex]::Replace($app, $pattern2, $replacement)
}

if ($fixed -eq $app) {
  Write-Host "WARNING: Did not find the broken setError line. Open frontend/src/App.tsx around your FREE_DAILY_LIMIT check." -ForegroundColor Yellow
} else {
  Write-Utf8NoBom -Path $appPath -Content $fixed
  Write-Host "✅ Patched App.tsx Free plan error line." -ForegroundColor Green
}

# --- 2) Fix .gitignore (stop ignoring frontend/src) ---
$gitignore = Join-Path $root ".gitignore"
if (Test-Path $gitignore) { Copy-Item $gitignore "$gitignore.bak-$ts" -Force }

$gi = @()
if (Test-Path $gitignore) { $gi = Get-Content $gitignore }

# Remove overly-broad ignores that break frontend/src
$gi = $gi | Where-Object { $_.Trim() -ne "src/" -and $_.Trim() -ne "src" }

# Ensure these ignores exist
$ensure = @(
  "",
  "# --- Confusion-AI ignores ---",
  ".env",
  ".env.*",
  "**/.env",
  "**/.env.*",
  "node_modules/",
  "**/node_modules/",
  "dist/",
  "**/dist/",
  ".vercel/",
  "*.log",
  "*.keystore",
  "*.jks"
)

foreach ($l in $ensure) {
  if (-not ($gi -contains $l)) { $gi += $l }
}

$gi | Set-Content -Path $gitignore -Encoding UTF8
Write-Host "✅ Updated .gitignore (no longer ignores frontend/src)." -ForegroundColor Green

# --- 3) Remove tracked node_modules/dist from Git index (critical for Vercel) ---
$pathsToUntrack = @(
  "frontend/node_modules",
  "frontend/dist",
  "node_modules",
  "dist"
)

foreach ($p in $pathsToUntrack) {
  $tracked = git ls-files $p 2>$null
  if ($tracked) {
    Write-Host "Untracking from git index: $p" -ForegroundColor Yellow
    git rm -r --cached $p | Out-Null
  }
}

# --- 4) Local build must pass ---
Push-Location $frontendPath
Write-Host "npm install --include=dev" -ForegroundColor Yellow
npm install --include=dev
if ($LASTEXITCODE -ne 0) { Pop-Location; Write-Error "npm install failed"; exit 1 }

Write-Host "npm run build" -ForegroundColor Yellow
npm run build
$exit = $LASTEXITCODE
Pop-Location

if ($exit -ne 0) {
  Write-Error "Build still failing. Paste the first ~30 lines of the build error."
  exit 1
}

Write-Host "✅ Local build succeeded." -ForegroundColor Green

# --- 5) Commit (on your current branch) ---
Write-Host "`nGit status:" -ForegroundColor Cyan
git status -sb

git add -A
$staged = git diff --cached --name-only
if (-not $staged) {
  Write-Host "No staged changes to commit." -ForegroundColor DarkGray
  exit 0
}

git commit -m "Fix build + remove node_modules/dist for Vercel"
Write-Host "✅ Commit created. Now push this branch." -ForegroundColor Cyan
Write-Host "Run: git push -u origin (git rev-parse --abbrev-ref HEAD)" -ForegroundColor Cyan
